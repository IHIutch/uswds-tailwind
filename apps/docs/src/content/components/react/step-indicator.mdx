export const stepIndicatorShortcut = `<StepIndicator.List>
  <StepIndicator.Segments />
</StepIndicator.List>`
export const stepIndicatorExpanded = `<StepIndicator.List>
  {({ steps }) =>
    steps.map(step => (
      <StepIndicator.ListItem key={step.label} status={step.status}>
        <StepIndicator.Segment status={step.status}>
          {step.label}
        </StepIndicator.Segment>
      </StepIndicator.ListItem>
    ))}
</StepIndicator.List>`

#### Segments shortcut

`StepIndicator.Segments` renders a `ListItem` and `Segment` for each step, passing the computed status to both. Inside `StepIndicator.List`, this:
<Code code={stepIndicatorShortcut} lang="tsx" />
is shorthand for the render function used in the full example above:
<Code code={stepIndicatorExpanded} lang="tsx" />
</div>
