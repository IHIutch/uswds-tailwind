export const inPageNavShortcut = `<InPageNav.List>
  <InPageNav.Items />
</InPageNav.List>`
export const inPageNavExpanded = `<InPageNav.List>
  {({ headings }) => headings.map(heading => (
    <InPageNav.Item key={heading.href}>
      <InPageNav.Link href={heading.href} depth={heading.depth}>
        {heading.label}
      </InPageNav.Link>
    </InPageNav.Item>
  ))}
</InPageNav.List>`

#### Items shortcut

`InPageNav.Items` renders an `Item` and `Link` for each heading. Inside `InPageNav.List`, this:
<Code code={inPageNavShortcut} lang="tsx" />
is shorthand for the render function used in the full example above:
<Code code={inPageNavExpanded} lang="tsx" />
