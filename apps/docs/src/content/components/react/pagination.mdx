  <Pagination.PrevTrigger />
  <Pagination.Pages />
  <Pagination.NextTrigger />
</Pagination.List>`
export const paginationExpanded = `<Pagination.List>
  {({ pages }) => (
    <>
      <Pagination.PrevTrigger />
      {pages.map((page, index) =>
        page.type === 'page'
          ? <Pagination.Item key={page.value} value={page.value} />
          : <Pagination.Ellipsis key={'ellipsis-' + index} />,
      )}
      <Pagination.NextTrigger />
    </>
  )}
</Pagination.List>`

#### Pages shortcut

`Pagination.Pages` renders a `Pagination.Item` for each visible page and a `Pagination.Ellipsis` for each gap. Inside `Pagination.List`, this:
<Code code={paginationShortcut} lang="tsx" />
is shorthand for the render function used in the full example above:
<Code code={paginationExpanded} lang="tsx" />
